cmake_minimum_required(VERSION 3.16)
project(n1_lvgl_radio VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(lvgl CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse-simple)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
find_package(Threads REQUIRED)
add_library(radio_core src/control.c src/config.c src/engine.c src/dsp.c src/rds.c)
target_include_directories(radio_core PUBLIC src)
target_link_libraries(radio_core PUBLIC PkgConfig::PULSE PkgConfig::JSONC Threads::Threads m)
target_compile_options(radio_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(n1-radio src/main.c src/window.c src/radio_font_20.c src/radio_font_extra_20.c src/radio_font_76.c)
target_link_libraries(n1-radio PRIVATE radio_core lvgl::lvgl)
target_compile_options(n1-radio PRIVATE -Wall -Wextra)
target_link_options(n1-radio PRIVATE "-Wl,--wrap=SDL_CreateWindow")
install(TARGETS n1-radio RUNTIME DESTINATION bin)
include(CTest)
