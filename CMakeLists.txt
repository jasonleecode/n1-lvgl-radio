cmake_minimum_required(VERSION 3.16)
project(n1_lvgl_radio VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(lvgl CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse-simple)
find_package(Threads REQUIRED)
add_library(radio_core src/config.c src/engine.c src/dsp.c)
target_include_directories(radio_core PUBLIC src)
target_link_libraries(radio_core PUBLIC PkgConfig::PULSE Threads::Threads m)
target_compile_options(radio_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(n1-radio src/main.c src/radio_font_20.c src/radio_font_extra_20.c src/radio_font_76.c)
target_link_libraries(n1-radio PRIVATE radio_core lvgl::lvgl)
target_compile_options(n1-radio PRIVATE -Wall -Wextra)
install(TARGETS n1-radio RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
 add_executable(radio-core-test tests/core-test.c)
 target_link_libraries(radio-core-test PRIVATE radio_core)
 target_compile_options(radio-core-test PRIVATE -UNDEBUG)
 add_test(NAME config-validation COMMAND radio-core-test)
 add_executable(radio-dsp-test tests/dsp-test.c)
 target_link_libraries(radio-dsp-test PRIVATE radio_core)
 target_compile_options(radio-dsp-test PRIVATE -UNDEBUG)
 add_test(NAME fm-antialias-filter COMMAND radio-dsp-test)
 add_executable(radio-hardware-check tests/hardware-check.c)
 target_link_libraries(radio-hardware-check PRIVATE radio_core)
 add_executable(radio-engine-test tests/engine-test.c)
 target_link_libraries(radio-engine-test PRIVATE radio_core)
 target_compile_options(radio-engine-test PRIVATE -UNDEBUG)
 add_test(NAME receiver-lifecycle COMMAND radio-engine-test)
 set_tests_properties(receiver-lifecycle PROPERTIES TIMEOUT 20)
endif()
